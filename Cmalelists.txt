cmake_minimum_required(VERSION 3.16)
project(lab_work_variant14 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Threads REQUIRED)

add_executable(work src/main.cpp)
target_link_libraries(work PRIVATE Threads::Threads)

if(MSVC)
  target_compile_options(work PRIVATE /W4 /utf-8)
else()
  target_compile_options(work PRIVATE -Wall -Wextra -pedantic)
endif()
